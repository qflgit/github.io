1. 记忆系统概述

1.1 为什么 Agent 需要记忆

LLM 的核心局限之一是无状态性：每次请求都是独立的，模型无法记住之前的对话或经历。记忆系统赋予 Agent 持续学习和适应的能力。

1.2 记忆系统的核心目标

  1. 持续性: 跨会话保持信息和经验

  2. 可检索性: 快速找到相关信息

  3. 可更新性: 动态修正和扩展知识

  4. 可压缩性: 在有限空间内保留关键信息

  5. 一致性: 避免记忆冲突和矛盾

  6. 隐私性: 保护敏感信息

class HybridMemoryStore:
    """混合存储架构，整合多种存储后端"""
    
    def __init__(self):
        # 短期记忆: Redis
        self.short_term = RedisStore(
            host='localhost',
            port=6379,
            ttl=3600  # 1小时过期
        )
        
        # 长期语义记忆: 向量数据库
        self.long_term_semantic = ChromaStore(
            collection_name='semantic_memory',
            embedding_model='text-embedding-3-large'
        )
        
        # 长期情景记忆: 时序数据库
        self.long_term_episodic = TimescaleDBStore(
            table_name='episodic_memory',
            retention_policy='1 year'
        )
        
        # 知识图谱: Neo4j
        self.knowledge_graph = Neo4jStore(
            uri='bolt://localhost:7687',
            database='knowledge'
        )
    
    async def store(self, memory: Memory, memory_type: str):
        """根据记忆类型选择存储后端"""
        
        if memory_type == 'short_term':
            await self.short_term.set(
                key=memory.id,
                value=memory.content,
                ttl=memory.ttl
            )
        
        elif memory_type == 'semantic':
            embedding = await self._embed(memory.content)
            await self.long_term_semantic.add(
                ids=memory.id,
                embeddings=embedding,
                metadatas=memory.metadata,
                documents=memory.content
            )
        
        elif memory_type == 'episodic':
            await self.long_term_episodic.insert(
                timestamp=memory.timestamp,
                event_type=memory.event_type,
                content=memory.content,
                context=memory.context
            )
        
        elif memory_type == 'knowledge_graph':
            await self.knowledge_graph.merge_entities(
                entities=memory.entities,
                relationships=memory.relationships
            )
    
    async def retrieve(
        self,
        query: str,
        memory_types: List[str] = None,
        top_k: int = 5
    ) -> List[Memory]:
        """多路检索，聚合结果"""
        memory_types = memory_types or ['semantic', 'episodic']
        
        results = []
        
        if 'semantic' in memory_types:
            embedding = await self._embed(query)
            semantic_results = await self.long_term_semantic.query(
                query_embeddings=embedding,
                n_results=top_k
            )
            results.extend(semantic_results)
        
        if 'episodic' in memory_types:
            episodic_results = await self.long_term_episodic.query(
                query=query,
                limit=top_k
            )
            results.extend(episodic_results)
        
        if 'knowledge_graph' in memory_types:
            kg_results = await self.knowledge_graph.query(
                query=query,
                limit=top_k
            )
            results.extend(kg_results)
        
        # 重排序
        return await self._rerank(results, query)

class SlidingWindowManager:
    """滑动窗口上下文管理器"""
    
    def __init__(self, max_tokens: int = 8000, keep_system: bool = True):
        self.max_tokens = max_tokens
        self.keep_system = keep_system
        self.messages = []
    
    def add_message(self, message: Dict) -> bool:
        """
        添加消息到上下文，如果超出限制则滑动窗口
        
        Returns: 是否成功添加
        """
        self.messages.append(message)
        
        # 检查是否超出限制
        total_tokens = self._count_tokens(self.messages)
        
        while total_tokens > self.max_tokens and len(self.messages) > 1:
            # 确定移除索引
            remove_idx = self._select_remove_index()
            removed = self.messages.pop(remove_idx)
            total_tokens = self._count_tokens(self.messages)
            print(f"[SlidingWindow] Removed message: {removed['role']}")
        
        return True
    
    def _select_remove_index(self) -> int:
        """选择要移除的消息索引"""
        # 保留系统消息（如果配置）
        start_idx = 0
        if self.keep_system and self.messages[0].get('role') == 'system':
            start_idx = 1
        
        # 策略1: 移除最旧的用户/助手消息对
        if len(self.messages) > start_idx + 2:
            return start_idx
        
        # 策略2: 如果只剩少量消息，移除最旧的
        return start_idx
    
    def _count_tokens(self, messages: List[Dict]) -> int:
        """估算 token 数量"""
        # 简化估算: 1 token ~ 4 字符
        total = 0
        for msg in messages:
            content = msg.get('content', '')
            total += len(content) // 4 + 1
        return total

class SummarizationManager:
    """摘要压缩上下文管理器"""
    
    def __init__(self, llm, max_tokens: int = 8000, summary_trigger: int = 6000):
        self.llm = llm
        self.max_tokens = max_tokens
        self.summary_trigger = summary_trigger
        self.messages = []
        self.summary = ""
    
    def add_message(self, message: Dict):
        """添加消息，必要时触发摘要"""
        self.messages.append(message)
        
        # 检查是否需要摘要
        total_tokens = self._count_tokens(self.messages)
        
        if total_tokens > self.summary_trigger:
            self._summarize_history()
    
    def _summarize_history(self):
        """对历史消息进行摘要"""
        # 保留最近的 N 条消息不摘要
        keep_recent = 4
        to_summarize = self.messages[:-keep_recent]
        keep_messages = self.messages[-keep_recent:]
        
        # 生成摘要
        history_text = self._format_messages(to_summarize)
        
        prompt = f"""
请将以下对话历史摘要为关键要点，保留重要事实、决策和上下文:


{history_text}


摘要（最多200字）:
"""
        
        new_summary = self.llm.generate(prompt)
        
        # 合并到现有摘要
        if self.summary:
            self.summary = f"{self.summary}\n\n新进展: {new_summary}"
        else:
            self.summary = new_summary
        
        # 替换消息历史
        self.messages = [
            {'role': 'system', 'content': f"历史摘要: {self.summary}"},
            *keep_messages
        ]
        
        print(f"[Summarization] History summarized. New length: {self._count_tokens(self.messages)} tokens")
    
    def get_context(self) -> List[Dict]:
        """获取当前上下文"""
        return self.messages

class HierarchicalSummaryManager:
    """分层摘要管理器 - 多级摘要"""
    
    def __init__(self, llm, levels: int = 3):
        self.llm = llm
        self.levels = levels
        self.summaries = {i: [] for i in range(levels)}
        self.current_level = 0
        self.messages_per_summary = 10
    
    def add_message(self, message: Dict):
        """添加消息到分层摘要系统"""
        
        # 添加到当前层级
        self.summaries[self.current_level].append(message)
        
        # 检查是否需要向上汇总
        if len(self.summaries[self.current_level]) >= self.messages_per_summary:
            self._promote_to_higher_level()
    
    def _promote_to_higher_level(self):
        """将当前层级摘要提升到更高层级"""
        
        for level in range(self.levels - 1):
            if len(self.summaries[level]) >= self.messages_per_summary:
                # 生成当前层级的摘要
                content = self._format_messages(self.summaries[level])
                summary = self._generate_summary(content, level)
                
                # 将摘要添加到下一层级
                self.summaries[level + 1].append({
                    'role': 'system',
                    'content': f"[L{level}摘要] {summary}",
                    'timestamp': time.time()
                })
                
                # 清空当前层级（保留最近的 2 条）
                self.summaries[level] = self.summaries[level][-2:]
    
    def get_context(self) -> List[Dict]:
        """获取分层上下文"""
        context = []
        
        # 从高到低收集摘要
        for level in range(self.levels - 1, -1, -1):
            if self.summaries[level]:
                if level > 0:
                    # 高级摘要作为系统消息
                    context.append({
                        'role': 'system',
                        'content': f"历史摘要 (Level {level}): {self._format_level_summary(level)}"
                    })
                else:
                    # 原始消息
                    context.extend(self.summaries[level])
        
        return context

        Args:
            content: 记忆内容
            memory_type: 记忆类型 (general/fact/experience/preference)
            metadata: 附加元数据
            memory_id: 可选的指定ID
            
        Returns:
            memory_id
        """
        memory_id = memory_id or str(uuid.uuid4())
        metadata = metadata or {}
        metadata['memory_type'] = memory_type
        metadata['timestamp'] = datetime.utcnow().isoformat()
        
        # 生成 embedding
        embedding = await self.embedding_model.aembed_query(content)
        
        # 存储到 ChromaDB
        self.collection.add(
            ids=[memory_id],
            embeddings=[embedding],
            metadatas=[metadata],
            documents=[content]
        )
        
        return memory_id
    
    async def retrieve(
        self,
        query: str,
        memory_type: str = None,
        top_k: int = 5,
        score_threshold: float = 0.7
    ) -> List[Dict]:
        """
        基于语义相似度检索记忆
        
        Args:
            query: 查询文本
            memory_type: 可选的记忆类型过滤
            top_k: 返回结果数量
            score_threshold: 相似度阈值
        """
        
        # 生成查询 embedding
        query_embedding = await self.embedding_model.aembed_query(query)
        
        # 构建过滤条件
        where_filter = {"memory_type": memory_type} if memory_type else None
        
        # 执行相似度搜索
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k,
            where=where_filter,
            include=["metadatas", "documents", "distances"]
        )
        
        # 格式化结果
        memories = []
        for i in range(len(results['ids'][0])):
            distance = results['distances'][0][i]
            similarity = 1 - distance  # 转换距离为相似度
            
            if similarity >= score_threshold:
                memories.append({
                    'id': results['ids'][0][i],
                    'content': results['documents'][0][i],
                    'metadata': results['metadatas'][0][i],
                    'similarity': similarity
                })
        
        return sorted(memories, key=lambda x: x['similarity'], reverse=True)
    
    async def update(
        self,
        memory_id: str,
        content: str = None,
        metadata: Dict = None
    ):
        """更新记忆"""
        if content:
            embedding = await self.embedding_model.aembed_query(content)
            self.collection.update(
                ids=[memory_id],
                embeddings=[embedding],
                documents=[content]
            )
        
        if metadata:
            self.collection.update(
                ids=[memory_id],
                metadatas=[metadata]
            )
    
    async def delete(self, memory_id: str):
        """删除记忆"""
        self.collection.delete(ids=[memory_id])

class MultiPathRetriever:
    """多路检索器 - 整合多种检索策略"""
    
    def __init__(self, stores: Dict[str, Any]):
        self.stores = stores
        self.retrieval_weights = {
            'semantic': 0.5,
            'keyword': 0.2,
            'temporal': 0.2,
            'graph': 0.1
        }
    
    async def retrieve(
        self,
        query: str,
        context: Dict = None,
        top_k: int = 10
    ) -> List[Dict]:
        """
        多路检索并融合结果
        
        Args:
            query: 查询文本
            context: 当前上下文（用于时间/空间过滤）
            top_k: 返回结果数量
        """
        
        # 并行执行多路检索
        retrieval_tasks = []
        
        # 语义检索
        if 'semantic' in self.stores:
            retrieval_tasks.append(
                self._semantic_retrieve(query, top_k * 2)
            )
        
        # 关键词检索
        if 'keyword' in self.stores:
            retrieval_tasks.append(
                self._keyword_retrieve(query, top_k * 2)
            )
        
        # 时序检索
        if 'temporal' in self.stores:
            retrieval_tasks.append(
                self._temporal_retrieve(query, context, top_k * 2)
            )
        
        # 图谱检索
        if 'graph' in self.stores:
            retrieval_tasks.append(
                self._graph_retrieve(query, top_k)
            )
        
        # 等待所有检索完成
        results = await asyncio.gather(*retrieval_tasks, return_exceptions=True)
        
        # 过滤异常结果
        valid_results = [r for r in results if not isinstance(r, Exception)]
        
        # 融合结果
        fused = self._fuse_results(valid_results, query)
        
        # 重排序
        reranked = await self._rerank(fused, query)
        
        return reranked[:top_k]
    
    def _fuse_results(
        self,
        results: List[List[Dict]],
        query: str
    ) -> List[Dict]:
        """融合多路检索结果"""
        
        # 收集所有结果并去重
        all_results = {}
        
        for result_list in results:
            for item in result_list:
                memory_id = item.get('id')
                
                if memory_id not in all_results:
                    all_results[memory_id] = item
                    all_results[memory_id]['sources'] = [item.get('source', 'unknown')]
                else:
                    # 合并分数（取最高）
                    all_results[memory_id]['score'] = max(
                        all_results[memory_id].get('score', 0),
                        item.get('score', 0)
                    )
                    all_results[memory_id]['sources'].append(
                        item.get('source', 'unknown')
                    )
        
        return list(all_results.values())
    
    async def _rerank(
        self,
        results: List[Dict],
        query: str
    ) -> List[Dict]:
        """重排序检索结果"""
        
        # 方法1: 使用交叉编码器
        # 方法2: 使用 LLM 评估相关性
        
        if len(results) <= 1:
            return results
        
        # 使用 LLM 评估相关性（简化版）
        scored_results = []
        for item in results:
            # 综合评分：原始分数 + 时效性 + 来源多样性
            base_score = item.get('score', 0)
            
            # 时效性加分（越新越好）
            age_hours = self._calculate_age(item)
            recency_boost = max(0, 1 - age_hours / 168)  # 一周内满分
            
            # 来源多样性加分
            source_count = len(set(item.get('sources', [])))
            diversity_boost = min(source_count * 0.1, 0.3)
            
            final_score = base_score * 0.6 + recency_boost * 0.3 + diversity_boost * 0.1
            
            scored_results.append({
                **item,
                'final_score': final_score
            })
        
        # 按最终分数排序
        scored_results.sort(key=lambda x: x['final_score'], reverse=True)
        
        return scored_results
    
    def _calculate_age(self, item: Dict) -> float:
        """计算记忆年龄（小时）"""
        timestamp = item.get('metadata', {}).get('timestamp')
        if not timestamp:
            return 9999
        
        from datetime import datetime
        try:
            created = datetime.fromisoformat(timestamp.replace('Z', '+00:00'))
            age = (datetime.utcnow() - created).total_seconds() / 3600
            return age
        except:
            return 9999

        self,

        existing_memories: List[Dict]
    ) -> List[Dict]:
        """
        检测新记忆与现有记忆的冲突

        1. 事实冲突: 同一实体不同属性值

        """

            )

                    existing
                )

                    conflicts.append({
                        'type': conflict_type,
                        'existing_memory': existing,
                        'new_memory': new_memory,
                        'similarity': similarity
                    })

        conflict: Dict
    ) -> Resolution:
        """解决记忆冲突"""
        
        resolution_strategy = conflict['type']
        
        if resolution_strategy == 'fact_conflict':
            # 事实冲突: 保留最新或置信度最高的
            return self._resolve_by_recency(conflict)
        
        elif resolution_strategy == 'time_conflict':
            # 时间冲突: 合并时间线
            return self._resolve_by_timeline(conflict)
        
        elif resolution_strategy == 'logical_conflict':
            # 逻辑冲突: 使用 LLM 推理解决
            return await self._resolve_by_reasoning(conflict)
        
        return Resolution(strategy='unknown', keep_both=True)

class MemoryJanitor:
    """记忆清理器 - 维护记忆健康度"""
    
    def __init__(
        self,
        store,
        ttl_config: Dict = None
    ):
        self.store = store
        self.ttl_config = ttl_config or {
            'short_term': 3600,      # 1小时
            'working': 86400,       # 1天
            'episodic': 2592000,    # 30天
            'semantic': None,        # 永久
            'procedural': None      # 永久
        }
    
    async def cleanup_expired(self):
        """清理过期记忆"""
        for memory_type, ttl in self.ttl_config.items():
            if ttl is not None:
                expired = await self.store.find_expired(
                    memory_type=memory_type,
                    max_age=ttl
                )
                
                for memory in expired:
                    # 检查是否需要归档
                    if memory.get('importance', 0) > 0.8:
                        await self._archive(memory)
                    else:
                        await self.store.delete(memory['id'])
    
    async def deduplicate(self):
        """去重相似记忆"""
        # 找到相似度 > 0.95 的记忆对
        duplicates = await self.store.find_duplicates(threshold=0.95)
        
        for pair in duplicates:
            # 合并或删除重复
            merged = await self._merge_memories(pair[0], pair[1])
            await self.store.update(pair[0]['id'], content=merged)
            await self.store.delete(pair[1]['id'])
    
    async def compress_old_memories(self, age_threshold_days: int = 30):
        """压缩旧记忆"""
        old_memories = await self.store.find_by_age(
            min_age_days=age_threshold_days
        )
        
        for memory in old_memories:
            if len(memory['content']) > 1000:
                compressed = await self._compress_memory(memory)
                await self.store.update(

class ConversationSummarizer:
    """对话历史摘要器"""
    
    def __init__(self, llm):
        self.llm = llm
    
    async def summarize(
        self,
        messages: List[Dict],
        summary_type: str = "progressive"
    ) -> str:
        """
        摘要对话历史
        
        Args:
            messages: 消息列表
            summary_type: progressive(渐进式)/final(最终式)/hierarchical(分层式)
        """
        
        if summary_type == 'progressive':
            return await self._progressive_summarize(messages)
        
        elif summary_type == 'final':
            return await self._final_summarize(messages)
        
        elif summary_type == 'hierarchical':
            return await self._hierarchical_summarize(messages)
    
    async def _progressive_summarize(self, messages: List[Dict]) -> str:
        """渐进式摘要 - 增量更新摘要"""
        
        # 分批处理，每批生成中间摘要
        batch_size = 10
        running_summary = ""
        
        for i in range(0, len(messages), batch_size):
            batch = messages[i:i+batch_size]
            batch_text = self._format_messages(batch)
            
            if running_summary:
                prompt = f"""
基于以下已有摘要和新对话，更新摘要:


已有摘要:
{running_summary}


新对话:
{batch_text}


更新后的摘要（保留关键信息，合并重复）:
"""
            else:
                prompt = f"""
摘要以下对话，提取关键信息、决策和待办事项:


{batch_text}


摘要:
"""
            
            running_summary = self.llm.generate(prompt)
        
        return running_summary
    
    async def _final_summarize(self, messages: List[Dict]) -> str:
        """最终式摘要 - 一次性摘要"""
        
        conversation_text = self._format_messages(messages)
        
        prompt = f"""
请对以下对话进行全面摘要，包含:
1. 主要话题和讨论内容
2. 关键决策和结论
3. 待办事项和下一步行动
4. 重要事实和数据
5. 用户偏好和意图


对话:
{conversation_text}


结构化摘要:
"""
        
        return self.llm.generate(prompt)
    
    async def _hierarchical_summarize(self, messages: List[Dict]) -> str:
        """分层摘要 - 多级摘要结构"""
        
        # 第一层: 按主题分组摘要
        topics = self._identify_topics(messages)
        topic_summaries = {}
        
        for topic, topic_messages in topics.items():
            summary = await self._summarize_topic(topic_messages)
            topic_summaries[topic] = summary
        
        # 第二层: 汇总所有主题
        overview = await self._summarize_topics(topic_summaries)
        
        return {
            'overview': overview,
            'topics': topic_summaries
        }
    
    def _format_messages(self, messages: List[Dict]) -> str:
        """格式化消息列表"""
        lines = []
        for msg in messages:
            role = msg.get('role', 'unknown')
            content = msg.get('content', '')
            lines.append(f"[{role}] {content}")
        return "\n".join(lines)

class SharedMemoryStore:
    """共享记忆存储 - 多 Agent 协作记忆"""
    
    def __init__(self, backend_store, access_control):
        self.backend = backend_store
        self.access_control = access_control
        self.subscribers = {}  # topic -> [agent_ids]
    
    async def share(
        self,
        agent_id: str,
        memory: Dict,
        share_level: str = "team",
        topics: List[str] = None
    ):
        """
        Agent 分享记忆到共享空间
        
        Args:
            share_level: public(公开)/team(团队)/private(私有)
            topics: 订阅主题列表
        """
        
        # 添加共享元数据
        memory['shared_by'] = agent_id
        memory['share_level'] = share_level
        memory['shared_at'] = datetime.utcnow().isoformat()
        memory['topics'] = topics or []
        
        # 存储到共享后端
        memory_id = await self.backend.store(memory)
        
        # 通知订阅者
        if topics:
            for topic in topics:
                await self._notify_subscribers(topic, memory)
        
        return memory_id
    
    async def retrieve_shared(
        self,
        agent_id: str,
        query: str,
        topics: List[str] = None,
        access_level: str = "read"
    ) -> List[Dict]:
        """
        检索共享记忆
        
        Args:
            access_level: read(只读)/write(读写)/admin(管理)
        """
        
        # 检查权限
        if not self.access_control.can_access(agent_id, access_level):
            raise AccessDeniedError(agent_id)
        
        # 构建过滤条件
        filters = {
            'share_level': {'$in': ['public', 'team']},
            '$or': [
                {'topics': {'$in': topics or []}},
                {'shared_by': agent_id}
            ]
        }
        
        # 执行检索
        results = await self.backend.retrieve(
            query=query,
            filters=filters
        )
        
        return results
    
    def subscribe(self, agent_id: str, topic: str):
        """订阅主题"""
        if topic not in self.subscribers:
            self.subscribers[topic] = set()
        self.subscribers[topic].add(agent_id)
    
    async def _notify_subscribers(self, topic: str, memory: Dict):
        """通知订阅者"""
        subscribers = self.subscribers.get(topic, set())
        
        for agent_id in subscribers:
            # 发送通知到 Agent
            await self._send_notification(
                agent_id,
                {
                    'type': 'new_shared_memory',
                    'topic': topic,
                    'memory_id': memory.get('id'),
                    'preview': memory.get('content', '')[:100]
                }
            )

class MemGPTMemory:
    """
    MemGPT 风格的分层记忆管理
    
    架构:
    - 主上下文 (Main Context): LLM 的直接上下文窗口
    - 外部上下文 (External Context): 超出窗口的记忆存储
    - 系统指令 (System Instructions): 控制记忆管理的规则
    """
    
    def __init__(
        self,
        llm,
        context_window_size: int = 8000,
        external_store = None
    ):
        self.llm = llm
        self.context_window_size = context_window_size
        self.external_store = external_store or VectorMemoryStore()

        self.system_instructions = []
        self.working_context = []
        self.fifo_queue = []
    
    async def process_message(self, message: str) -> str:
        """
        处理消息，管理上下文溢出
        
        流程:
        1. 将消息添加到工作上下文
        2. 检查上下文溢出
        3. 如果溢出，将旧内容归档到外部存储
        4. 执行 LLM 推理
        5. 将响应添加到工作上下文
        """
        
        # 添加消息到工作上下文
        self.working_context.append({'role': 'user', 'content': message})
        
        # 检查并处理溢出
        await self._manage_overflow()
        
        # 构建完整上下文
        context = self._build_context()
        
        # 执行推理
        response = await self.llm.generate(context)
        
        # 添加响应到工作上下文
        self.working_context.append({'role': 'assistant', 'content': response})
        
        return response
    
    async def _manage_overflow(self):
        """管理上下文溢出"""
        
        while self._estimate_tokens() > self.context_window_size:
            # 找到最旧的可归档内容
            to_archive = self._find_oldest_archivable()
            
            if to_archive:
                # 归档到外部存储
                await self.external_store.store(
                    content=to_archive['content'],
                    memory_type='archived_conversation',
                    metadata={'role': to_archive['role']}
                )
                
                # 从主上下文移除
                self.working_context.remove(to_archive)
            else:
                # 无法归档，进行摘要
                await self._summarize_oldest()
    
    def _build_context(self) -> List[Dict]:
        """构建完整上下文"""
        return [
            *self.system_instructions,
            *self.working_context
        ]

        else:
            return {'M': 128, 'ef_construction': 800, 'ef_search': 512}
    
    def get_ivf_config(self, data_size: int) -> Dict:
        """
        IVF 索引配置（适合大数据量）
        
        IVF 参数:
        - nlist: 聚类中心数量
        - nprobe: 查询时搜索的聚类数
        """
        
        nlist = min(4 * int(data_size ** 0.5), 65536)
        nprobe = max(nlist // 100, 10)
        
        return {'nlist': nlist, 'nprobe': nprobe}

from functools import lru_cache
import hashlib


class MemoryCache:
    """记忆缓存系统"""
    
    def __init__(self, max_size: int = 1000, ttl: int = 300):
        self.max_size = max_size
        self.ttl = ttl
        self.cache = {}
        self.timestamps = {}
    
    def _generate_key(self, query: str, filters: Dict = None) -> str:
        """生成缓存键"""
        key_data = f"{query}:{json.dumps(filters, sort_keys=True) if filters else ''}"
        return hashlib.md5(key_data.encode()).hexdigest()
    
    def get(self, query: str, filters: Dict = None) -> Optional[List[Dict]]:
        """获取缓存"""
        key = self._generate_key(query, filters)
        
        if key in self.cache:
            # 检查 TTL
            if time.time() - self.timestamps[key] < self.ttl:
                return self.cache[key]
            else:
                # 过期，移除
                del self.cache[key]
                del self.timestamps[key]
        
        return None
    
    def set(self, query: str, results: List[Dict], filters: Dict = None):
        """设置缓存"""
        key = self._generate_key(query, filters)
        
        # 检查容量
        if len(self.cache) >= self.max_size:
            # 移除最旧的
            oldest_key = min(self.timestamps, key=self.timestamps.get)
            del self.cache[oldest_key]
            del self.timestamps[oldest_key]
        
        self.cache[key] = results
        self.timestamps[key] = time.time()

# 最小可运行的记忆系统
from typing import List, Dict
import json


class SimpleMemory:
    """简单记忆系统 - 内存存储 + 关键词检索"""
    
    def __init__(self):
        self.memories = []
    
    def add(self, content: str, metadata: Dict = None):
        """添加记忆"""
        self.memories.append({
            'id': len(self.memories),
            'content': content,
            'metadata': metadata or {},
            'keywords': set(content.lower().split())
        })
    
    def search(self, query: str, top_k: int = 3) -> List[Dict]:
        """关键词检索"""
        query_words = set(query.lower().split())
        
        scored = []
        for memory in self.memories:
            overlap = len(query_words & memory['keywords'])
            if overlap > 0:
                scored.append((overlap, memory))
        
        scored.sort(key=lambda x: x[0], reverse=True)
        return [m for _, m in scored[:top_k]]


# 使用
memory = SimpleMemory()
memory.add("Apple 2025年营收3910亿美元", {'type': 'fact'})
memory.add("Google 2025年营收3500亿美元", {'type': 'fact'})
results = memory.search("Apple 营收")
for r in results:
    print(r['content'])
