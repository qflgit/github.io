2. 协作理论基础

2.1 多 Agent 系统 (MAS) 理论

多 Agent 系统 (Multi-Agent Systems, MAS) 是分布式人工智能的核心分支，研究多个自主 Agent 如何通过交互协作解决复杂问题。

核心概念:

  • 自主性 (Autonomy): Agent 独立决策的能力

  • 社会性 (Social Ability): Agent 间通信交互的能力

  • 反应性 (Reactivity): 感知环境并响应的能力

  • 主动性 (Pro-activeness): 主动追求目标的能力

3. 经典协作架构模式

3.1 层级式架构 (Hierarchical)

核心思想: 一个管理者 Agent 负责任务分解和分配，多个工作者 Agent 执行具体任务。

xxxxxxxxxx
                    +----------------+
                    |   管理者 Agent  |
                    |  (Manager)     |
                    | - 任务分解      |
                    | - 分配调度      |
                    | - 结果汇总      |
                    +--------+-------+
                             |
            +----------------+----------------+
            |                |                |
    +-------+-------+ +------+------+ +------+------+
    | 工作者 Agent A | | 工作者 Agent B | | 工作者 Agent C |
    | - 数据收集      | | - 数据分析      | | - 报告生成      |
    +---------------+ +-------------+ +-------------+

适用场景: 项目管理、工作流编排、任务分发

优势: 结构清晰、责任明确、易于控制

劣势: 单点瓶颈、管理者负载重、灵活性差

代码示例:

xxxxxxxxxx
class ManagerAgent:
    """管理者 Agent，负责任务分解和分配"""
    
    def __init__(self, workers: List['WorkerAgent']):
        self.workers = {w.role: w for w in workers}
    
    def execute(self, task: str) -> str:
        """执行复杂任务"""
        # 步骤1: 分解任务
        subtasks = self._decompose_task(task)
        
        # 步骤2: 分配任务
        results = {}
        for subtask in subtasks:
            worker = self._select_worker(subtask)
            result = worker.execute(subtask)
            results[subtask['id']] = result
        
        # 步骤3: 汇总结果
        final = self._aggregate_results(results)
        return final
    
    def _decompose_task(self, task: str) -> List[Dict]:
        """将任务分解为子任务"""
        # 使用 LLM 进行任务分解
        prompt = f"""
将以下任务分解为3-5个子任务，每个子任务包含:
- id: 子任务编号
- role: 执行角色 (researcher/analyst/writer)
- description: 详细描述
- dependencies: 依赖的子任务id列表


任务: {task}
"""
        return self.llm.generate(prompt)
    
    def _select_worker(self, subtask: Dict) -> 'WorkerAgent':
        """根据子任务选择合适的工作者"""
        role = subtask['role']
        return self.workers.get(role)
    
    def _aggregate_results(self, results: Dict) -> str:
        """汇总所有子任务结果"""
        prompt = f"""
基于以下子任务结果，生成最终答案:
{results}
"""
        return self.llm.generate(prompt)


class WorkerAgent:
    """工作者 Agent，执行具体子任务"""
    
    def __init__(self, role: str, llm):
        self.role = role
        self.llm = llm
    
    def execute(self, subtask: Dict) -> str:
        """执行子任务"""
        prompt = f"""
作为{self.role}，执行以下任务:
{subtask['description']}
"""
        return self.llm.generate(prompt)

适用场景: 头脑风暴、分布式问题求解、协商谈判

3.3 流水线架构 (Pipeline)

核心思想: Agent 按固定顺序连接，数据依次流经每个 Agent 进行处理。

xxxxxxxxxx
Input --> [Agent A] --> [Agent B] --> [Agent C] --> Output
           (提取)       (分析)       (生成)

适用场景: 数据处理、内容生成、ETL 流程

优势: 流程清晰、易于调试、模块化程度高

劣势: 灵活性差、单点阻塞、难以并行

代码示例:

xxxxxxxxxx
class Pipeline:
    """流水线协作模式"""
    
    def __init__(self):
        self.stages = []
    
    def add_stage(self, agent, name: str):
        """添加流水线阶段"""
        self.stages.append({'agent': agent, 'name': name})
    
    def execute(self, input_data: str) -> str:
        """执行流水线"""
        data = input_data
        
        for stage in self.stages:
            print(f"[Pipeline] Stage: {stage['name']}")
            data = stage['agent'].process(data)
        
        return data


# 使用示例
pipeline = Pipeline()
pipeline.add_stage(DataExtractorAgent(), "提取")
pipeline.add_stage(DataAnalyzerAgent(), "分析")
pipeline.add_stage(ReportGeneratorAgent(), "生成")
result = pipeline.execute("原始数据")

class DirectCommunication:
    """点对点直接通信"""
    
    def __init__(self):
        self.agents = {}
        self.message_queue = {}
    
    def register_agent(self, agent_id: str, agent):
        self.agents[agent_id] = agent
        self.message_queue[agent_id] = asyncio.Queue()
    
    async def send_message(
        self,
        sender_id: str,
        recipient_id: str,
        message: Dict
    ):
        """发送消息到指定 Agent"""
        if recipient_id not in self.agents:
            raise ValueError(f"Agent {recipient_id} not found")
        
        envelope = {
            'message_id': str(uuid.uuid4()),
            'timestamp': datetime.utcnow().isoformat(),
            'sender': sender_id,
            'recipient': recipient_id,
            'content': message
        }
        
        await self.message_queue[recipient_id].put(envelope)
    
    async def receive_message(self, agent_id: str) -> Dict:
        """接收消息"""
        return await self.message_queue[agent_id].get()

4.3.2 广播通信 (Broadcast)

xxxxxxxxxx
class BroadcastCommunication:
    """广播通信"""
    
    def __init__(self):
        self.subscribers = {}
    
    def subscribe(self, topic: str, agent_id: str, callback):
        """订阅主题"""
        if topic not in self.subscribers:
            self.subscribers[topic] = {}
        self.subscribers[topic][agent_id] = callback
    
    def unsubscribe(self, topic: str, agent_id: str):
        """取消订阅"""
        if topic in self.subscribers:
            self.subscribers[topic].pop(agent_id, None)
    
    async def broadcast(
        self,
        topic: str,
        message: Dict,
        exclude: List[str] = None
    ):
        """广播消息到所有订阅者"""
        exclude = exclude or []
        
        if topic not in self.subscribers:
            return
        
        tasks = []
        for agent_id, callback in self.subscribers[topic].items():
            if agent_id not in exclude:
                tasks.append(callback(message))
        
        await asyncio.gather(*tasks, return_exceptions=True)

class BlackboardCommunication:
    """黑板通信系统"""
    
    def __init__(self):
        self.data = {}
        self.locks = {}
        self.subscribers = {}
    
    async def write(
        self,
        key: str,
        value: Any,
        agent_id: str
    ):
        """写入黑板"""
        async with self.locks.get(key, asyncio.Lock()):
            self.data[key] = {
                'value': value,
                'writer': agent_id,
                'timestamp': datetime.utcnow().isoformat(),
                'version': self.data.get(key, {}).get('version', 0) + 1
            }
            
            # 通知订阅者
            if key in self.subscribers:
                for callback in self.subscribers[key]:
                    await callback(key, value, agent_id)
    
    async def read(self, key: str) -> Any:
        """读取黑板"""
        entry = self.data.get(key)
        return entry['value'] if entry else None
    
    def subscribe(self, key: str, callback):
        """订阅数据变化"""
        if key not in self.subscribers:
            self.subscribers[key] = []
        self.subscribers[key].append(callback)

class CapabilityBasedDecomposition:
    """基于 Agent 能力的任务分解"""
    
    def __init__(self, agents: List['Agent']):
        self.agents = agents
        self.capability_map = self._build_capability_map()
    
    def _build_capability_map(self) -> Dict[str, List[str]]:
        """构建能力到 Agent 的映射"""
        mapping = {}
        for agent in self.agents:
            for cap in agent.capabilities:
                if cap not in mapping:
                    mapping[cap] = []
                mapping[cap].append(agent.id)
        return mapping
    
    def decompose(self, task: str) -> List[SubTask]:
        """
        将任务分解为匹配 Agent 能力的子任务
        """
        # 分析任务所需能力
        required_capabilities = self._analyze_task(task)
        
        subtasks = []
        for i, cap in enumerate(required_capabilities):
            capable_agents = self.capability_map.get(cap, [])
            
            subtasks.append(SubTask(
                id=f"subtask-{i}",
                capability=cap,
                description=f"执行 {cap} 相关任务",
                candidate_agents=capable_agents,
                priority=i
            ))
        
        return subtasks
    
    def _analyze_task(self, task: str) -> List[str]:
        """分析任务所需能力"""
        prompt = f"""
分析以下任务需要哪些能力，返回能力列表:
{task}
"""
        return self.llm.generate(prompt).split(',')

class DependencyBasedDecomposition:
    """基于依赖关系的任务分解"""
    
    def decompose(self, task: str) -> DAG:
        """
        将任务分解为 DAG，识别并行和串行关系
        """
        prompt = f"""
将以下任务分解为有向无环图(DAG):
任务: {task}


要求:
1. 识别所有子任务
2. 明确子任务间的依赖关系
3. 标注哪些子任务可以并行执行
4. 返回 JSON 格式: {nodes: [...], edges: [...]}
"""
        
        dag_json = self.llm.generate(prompt)
        return DAG.from_json(dag_json)

class LoadBalancingAllocator:
    """负载均衡任务分配器"""
    
    def __init__(self, agents: List['Agent']):
        self.agents = agents
        self.load_tracker = {a.id: 0 for a in agents}
    
    def allocate(self, subtasks: List[SubTask]) -> Dict[str, str]:
        """
        将子任务分配给负载最轻的 Agent
        
        Returns: {subtask_id: agent_id}
        """
        allocation = {}
        
        for subtask in subtasks:
            # 找到有能力的 Agent 中负载最轻的
            capable_agents = [
                a for a in self.agents
                if subtask.capability in a.capabilities
            ]
            
            if not capable_agents:
                raise NoCapableAgentError(subtask.capability)
            
            # 选择负载最轻的
            best_agent = min(
                capable_agents,
                key=lambda a: self.load_tracker[a.id]
            )
            
            allocation[subtask.id] = best_agent.id
            self.load_tracker[best_agent.id] += 1
        
        return allocation

class AuctionAllocator:
    """拍卖机制任务分配器"""
    
    def __init__(self, agents: List['Agent']):
        self.agents = agents
    
    async def allocate(self, subtask: SubTask) -> str:
        """
        通过拍卖将任务分配给 Agent
        
        流程:
        1. 发布任务
        2. 收集出价
        3. 评估出价
        4. 选择赢家
        5. 签订合同
        """
        
        # 步骤1: 发布任务
        announcement = {
            'task': subtask,
            'deadline': time.time() + 10  # 10秒出价时间
        }
        
        # 步骤2: 收集出价
        bids = []
        for agent in self.agents:
            if subtask.capability in agent.capabilities:
                bid = await agent.bid(announcement)
                if bid:
                    bids.append((agent.id, bid))
        
        if not bids:
            raise NoBidsError()
        
        # 步骤3: 评估出价（综合考虑价格、能力、历史表现）
        scored_bids = [
            (agent_id, self._score_bid(bid))
            for agent_id, bid in bids
        ]
        
        # 步骤4: 选择赢家（得分最高）
        winner_id, _ = max(scored_bids, key=lambda x: x[1])
        
        return winner_id
    
    def _score_bid(self, bid: Dict) -> float:
        """评估出价"""
        # 综合评分：价格(越低越好)、能力(越高越好)、成功率(越高越好)
        price_score = 1.0 / (bid['price'] + 1)
        capability_score = bid['capability_level'] / 10
        success_score = bid['historical_success_rate']
        
        return price_score * 0.3 + capability_score * 0.3 + success_score * 0.4

class VotingConsensus:
    """投票共识机制"""
    
    def __init__(self, agents: List['Agent'], threshold: float = 0.6):
        self.agents = agents
        self.threshold = threshold
    
    async def reach_consensus(self, proposal: str) -> Dict:
        """
        通过投票达成共识
        
        Returns: {'consensus': bool, 'result': str, 'votes': Dict}
        """
        
        # 收集投票
        votes = {}
        for agent in self.agents:
            vote = await agent.vote(proposal)
            votes[agent.id] = vote
        
        # 统计结果
        yes_votes = sum(1 for v in votes.values() if v == 'yes')
        total_votes = len(votes)
        ratio = yes_votes / total_votes
        
        consensus_reached = ratio >= self.threshold
        
        return {
            'consensus': consensus_reached,
            'result': 'accepted' if consensus_reached else 'rejected',
            'votes': votes,
            'ratio': ratio,
            'yes_count': yes_votes,
            'no_count': total_votes - yes_votes
        }

class NegotiationConsensus:
    """协商共识机制"""
    
    async def negotiate(
        self,
        agents: List['Agent'],
        issue: str,
        max_rounds: int = 5
    ) -> Dict:
        """
        多轮协商达成共识
        
        流程:
        1. 各方提出初始立场
        2. 交换意见，识别分歧
        3. 提出妥协方案
        4. 评估接受度
        5. 重复直到达成共识或达到最大轮数
        """
        
        # 初始立场
        positions = {}
        for agent in agents:
            positions[agent.id] = await agent.propose_position(issue)
        
        for round_num in range(max_rounds):
            print(f"[Negotiation] Round {round_num + 1}")
            
            # 检查是否已达成共识
            if self._check_consensus(positions):
                return {
                    'consensus': True,
                    'rounds': round_num + 1,
                    'final_position': list(positions.values())[0],
                    'positions': positions
                }
            
            # 协商：每个 Agent 根据他人立场调整
            new_positions = {}
            for agent in agents:
                others_positions = {
                    k: v for k, v in positions.items()
                    if k != agent.id
                }
                new_positions[agent.id] = await agent.negotiate(
                    positions[agent.id],
                    others_positions
                )
            
            positions = new_positions
        
        # 达到最大轮数，未达成共识
        return {
            'consensus': False,
            'rounds': max_rounds,
            'positions': positions,
            'message': 'Max rounds reached without consensus'
        }
    
    def _check_consensus(self, positions: Dict) -> bool:
        """检查是否达成共识"""
        unique_positions = set(str(v) for v in positions.values())
        return len(unique_positions) == 1

        self,
        conflicts: List[Conflict],
        priority_rules: List[Callable]
    ) -> Resolution:
        """
        基于优先级规则解决冲突
        
        优先级规则（按顺序应用）:
        1. 角色优先级（管理者 > 工作者）

        """

            winner = rule(conflicts)
            if winner:
                return Resolution(
                    winner=winner,
                    losers=[c for c in conflicts if c != winner],
                    strategy='priority',
                    reason=f'Won by priority rule: {rule.__name__}'
                )
        
        # 无法通过优先级解决，使用协商
        return self._fallback_to_negotiation(conflicts)

class VotingResolution:
    """投票解决冲突"""
    
    async def resolve(
        self,
        conflict: Conflict,
        voters: List['Agent']
    ) -> Resolution:
        """
        通过投票解决冲突
        
        适用于:
        - 信息冲突（哪个数据源更可靠）
        - 方案选择（多个方案选哪个）
        """
        
        options = conflict.options
        
        # 收集投票
        votes = {option: 0 for option in options}
        for voter in voters:
            vote = await voter.vote(options, conflict)
            votes[vote] += 1
        
        # 选择得票最多
        winner = max(votes, key=votes.get)
        
        return Resolution(
            winner=winner,
            votes=votes,
            strategy='voting',
            total_votes=len(voters)
        )

9. A2A 协议详解

9.1 什么是 A2A

A2A (Agent-to-Agent Protocol) 是 Google 于 2025 年推出的开放协议，用于标准化不同 Agent 之间的通信和协作。

核心定位: A2A 是 Agent 世界的 "SMTP" - 它定义了 Agent 间邮件式的通信标准，使得来自不同厂商、不同框架的 Agent 可以互操作。

9.2 A2A 与 MCP 的关系

xxxxxxxxxx
MCP (Model Context Protocol)          A2A (Agent-to-Agent Protocol)
    |                                      |    |-- Agent <-> 工具/数据                 |-- Agent <-> Agent    |-- 垂直集成                            |-- 水平协作    |-- 获取能力                            |-- 协作完成任务    |-- 类比：HTTP                          |-- 类比：SMTP/XMPP    |                                      |    +-------------- 互补关系 ---------------+                  |                  v         完整 Agent 生态系统```


### 9.3 A2A 核心概念


| 概念 | 说明 | 类比 |
|------|------|------|
| **Agent Card** | Agent 的能力描述和元数据 | 名片/简历 |
| **Task** | 协作任务单元 | 邮件主题 |
| **Message** | 任务中的具体消息 | 邮件正文 |
| **Artifact** | 任务产出的成果 | 附件 |
| **Skill** | Agent 的具体能力 | 技能标签 |
| **Push Notification** | 实时通知机制 | 推送通知 |

class DynamicTeamFormation:
    """动态团队组建系统"""
    
    def __init__(self, agent_pool: List['Agent']):
        self.agent_pool = agent_pool
        self.capability_matrix = self._build_capability_matrix()
    
    def form_team(self, task: str, team_size: int = 3) -> Team:
        """
        根据任务需求动态组建团队
        
        策略:
        1. 分析任务所需能力
        2. 匹配具有相关能力的 Agent
        3. 考虑 Agent 间的协作历史
        4. 平衡团队多样性
        """
        
        # 分析任务需求
        required_caps = self._analyze_task_requirements(task)
        
        # 评分并排序候选 Agent
        scored_agents = []
        for agent in self.agent_pool:
            score = self._calculate_utility(agent, required_caps)
            scored_agents.append((agent, score))
        
        scored_agents.sort(key=lambda x: x[1], reverse=True)
        
        # 选择 top-k，确保能力覆盖
        selected = []
        covered_caps = set()
        
        for agent, score in scored_agents:
            if len(selected) >= team_size:
                break
            
            agent_caps = set(agent.capabilities)
            new_caps = agent_caps - covered_caps
            
            # 优先选择能覆盖新能力的 Agent
            if new_caps or len(selected) < team_size // 2:
                selected.append(agent)
                covered_caps.update(agent_caps)
        
        return Team(members=selected, capabilities=covered_caps)
    
    def _calculate_utility(
        self,
        agent: 'Agent',
        required_caps: List[str]
    ) -> float:
        """计算 Agent 对任务的效用分数"""
        
        # 能力匹配度
        matching_caps = set(agent.capabilities) & set(required_caps)
        match_score = len(matching_caps) / len(required_caps)
        
        # 历史成功率
        success_score = agent.historical_success_rate
        
        # 当前负载（越低越好）
        load_score = 1.0 / (1 + agent.current_load)
        
        # 协作评分
        collaboration_score = agent.collaboration_rating
        
        return (
            match_score * 0.4 +
            success_score * 0.3 +
            load_score * 0.2 +
            collaboration_score * 0.1
        )

class BatchedCommunication:
    """批量通信优化器"""
    
    def __init__(self, max_batch_size: int = 10, max_wait_ms: int = 100):
        self.max_batch_size = max_batch_size
        self.max_wait_ms = max_wait_ms
        self.pending_messages = []
        self.lock = asyncio.Lock()
    
    async def send(self, message: Dict):
        """发送消息（自动批量）"""
        async with self.lock:
            self.pending_messages.append(message)
            
            if len(self.pending_messages) >= self.max_batch_size:
                await self._flush()
    
    async def _flush(self):
        """刷新批量消息"""
        if not self.pending_messages:
            return
        
        batch = self.pending_messages[:self.max_batch_size]
        self.pending_messages = self.pending_messages[self.max_batch_size:]
        
        # 压缩并发送
        compressed = self._compress_batch(batch)
        await self._send_batch(compressed)
    
    def _compress_batch(self, messages: List[Dict]) -> bytes:
        """压缩消息批次"""
        import json
        import zlib
        
        json_bytes = json.dumps(messages).encode('utf-8')
        return zlib.compress(json_bytes)

class ContextCompressor:
    """上下文压缩器，减少多 Agent 对话的 Token 消耗"""
    
    def __init__(self, llm, max_context_length: int = 8000):
        self.llm = llm
        self.max_context_length = max_context_length
    
    def compress(self, messages: List[Dict]) -> List[Dict]:
        """
        压缩消息历史，保留关键信息
        
        策略:
        1. 保留系统消息和最近 N 轮对话
        2. 对早期对话进行摘要
        3. 提取关键决策和结论
        """
        
        total_length = sum(len(str(m)) for m in messages)
        
        if total_length <= self.max_context_length:
            return messages
        
        # 分离系统消息、近期消息和早期消息
        system_messages = [m for m in messages if m.get('role') == 'system']
        recent_messages = messages[-5:]  # 保留最近5轮
        early_messages = messages[len(system_messages):-5]
        
        # 对早期消息进行摘要
        if early_messages:
            summary = self._summarize_messages(early_messages)
            compressed = (
                system_messages +
                [{'role': 'system', 'content': f'历史摘要: {summary}'}] +
                recent_messages
            )
        else:
            compressed = system_messages + recent_messages
        
        return compressed
    
    def _summarize_messages(self, messages: List[Dict]) -> str:
        """摘要消息历史"""
        content = '\n'.join([m.get('content', '') for m in messages])
        
        prompt = f"""
请将以下对话历史摘要为关键要点（最多5条）:
{content}
"""
        
        return self.llm.generate(prompt)

        return '\n'.join(mermaid)
    
    def analyze_performance(self) -> Dict:
        """分析性能指标"""
        if not self.message_log:
            return {}
        
        start_time = self.message_log[0]['timestamp']
        end_time = self.message_log[-1]['timestamp']
        total_duration = end_time - start_time
        
        message_count = len(self.message_log)
        
        # 计算各 Agent 的参与度
        agent_activity = {}
        for entry in self.message_log:
            sender = entry['message'].get('sender', 'unknown')
            agent_activity[sender] = agent_activity.get(sender, 0) + 1
        
        return {
            'total_duration': total_duration,
            'message_count': message_count,
            'messages_per_second': message_count / total_duration if total_duration > 0 else 0,
            'agent_activity': agent_activity,
            'most_active_agent': max(agent_activity, key=agent_activity.get) if agent_activity else None
        }

# 最小可运行的多 Agent 协作系统
from typing import List, Dict
import asyncio


class SimpleAgent:
    def __init__(self, name: str, role: str):
        self.name = name
        self.role = role
    
    async def process(self, task: str) -> str:
        return f"[{self.name}] 处理完成: {task}"


class SimpleCollaboration:
    def __init__(self, agents: List[SimpleAgent]):
        self.agents = agents
    
    async def run(self, task: str) -> List[str]:
        """并行执行所有 Agent"""
        tasks = [agent.process(task) for agent in self.agents]
        return await asyncio.gather(*tasks)


# 使用
agents = [
    SimpleAgent('研究员', 'research'),
    SimpleAgent('分析师', 'analysis'),
    SimpleAgent('作家', 'writing')
]


collab = SimpleCollaboration(agents)
results = asyncio.run(collab.run('分析AI趋势'))
for r in results:
    print(r)
