1. 工具调用概述

1.1 什么是工具调用

工具调用(Tool Calling) 是 Agent 与外部世界交互的核心机制。它允许 LLM 在推理过程中主动调用外部函数、API、数据库查询等工具,获取实时信息或执行具体操作,从而突破模型内部知识的局限。

核心定义: 工具调用是 Agent 将自然语言意图转化为结构化函数调用,执行外部操作,并将结果反馈回推理循环的完整流程。

4.1.1 基于装饰器的注册

from typing import Annotated
from pydantic import Field


class ToolRegistry:
    def __init__(self):
        self._tools = {}
        self._schemas = []


    def register(self, func, name=None, description=None):
        tool_name = name or func.__name__
        tool_desc = description or func.__doc__
        
        # 自动生成 Schema
        schema = self._generate_schema(func, tool_name, tool_desc)
        
        self._tools[tool_name] = {
            'function': func,
            'schema': schema,
            'metadata': {
                'name': tool_name,
                'description': tool_desc,
                'version': '1.0.0',
                'tags': []
            }
        }
        self._schemas.append(schema)
        return func


    def _generate_schema(self, func, name, description):
        # 使用 inspect 和 typing 自动生成 JSON Schema
        import inspect
        sig = inspect.signature(func)
        
        properties = {}
        required = []
        
        for param_name, param in sig.parameters.items():
            if param_name == 'self':
                continue
                
            param_info = {'type': 'string'}  # 简化处理
            
            # 处理 Annotated 类型
            if hasattr(param.annotation, '__metadata__'):
                for meta in param.annotation.__metadata__:
                    if isinstance(meta, Field):
                        param_info['description'] = meta.description
                        if meta.default is not None:
                            param_info['default'] = meta.default
                        else:
                            required.append(param_name)
            
            properties[param_name] = param_info
        
        return {
            'type': 'function',
            'function': {
                'name': name,
                'description': description,
                'parameters': {
                    'type': 'object',
                    'properties': properties,
                    'required': required
                }
            }
        }


# 使用示例
registry = ToolRegistry()


@registry.register
def web_search(
    query: Annotated[str, Field(description="搜索查询语句")],
    num_results: Annotated[int, Field(description="返回结果数量", default=5)] = 5
):
    """使用搜索引擎查找信息"""
    # 实现搜索逻辑
    return search_engine.query(query, num_results)


@registry.register
def calculate(
    expression: Annotated[str, Field(description="数学表达式，如: 2 + 3 * 4")]
):
    """执行数学计算"""
    return eval(expression)

class ToolSelector:
    """智能工具选择器"""
    
    def __init__(self, llm, registry):
        self.llm = llm
        self.registry = registry
    
    def select_tools(
        self, 
        query: str, 
        available_tools: List[Dict],
        max_tools: int = 5
    ) -> List[Dict]:
        """
        基于查询意图选择最合适的工具
        
        策略1: 基于描述相似度 (Embedding 相似度)
        策略2: 基于历史成功率
        策略3: 基于 LLM 判断
        """
        
        # 策略1: Embedding 相似度排序
        query_embedding = self.llm.embed(query)
        
        scored_tools = []
        for tool in available_tools:
            tool_desc = tool['function']['description']
            tool_embedding = self.llm.embed(tool_desc)
            similarity = cosine_similarity(query_embedding, tool_embedding)
            
            # 结合历史成功率
            success_rate = tool.get('metadata', {}).get('success_rate', 0.8)
            score = similarity * 0.7 + success_rate * 0.3
            
            scored_tools.append((score, tool))
        
        # 按分数排序并返回 Top-K
        scored_tools.sort(key=lambda x: x[0], reverse=True)
        return [tool for _, tool in scored_tools[:max_tools]]

import json
from typing import Any, Dict, get_type_hints
from pydantic import BaseModel, ValidationError


class ParameterParser:
    """参数解析与验证器"""
    
    def __init__(self, tool_schema: Dict):
        self.schema = tool_schema
        self.parameters = tool_schema['function']['parameters']
    
    def parse(self, arguments_json: str) -> Dict[str, Any]:
        """
        解析并验证 LLM 生成的参数
        
        Args:
            arguments_json: LLM 生成的 JSON 字符串
            
        Returns:
            解析后的参数字典
            
        Raises:
            ParseError: 解析失败
            ValidationError: 参数不合法
        """
        
        # 步骤1: JSON 解析
        try:
            params = json.loads(arguments_json)
        except json.JSONDecodeError as e:
            raise ParseError(f"Invalid JSON: {e}")
        
        # 步骤2-4: 验证与转换
        validated = self._validate_parameters(params)
        
        return validated
    
    def _validate_parameters(self, params: Dict) -> Dict[str, Any]:
        """验证参数合法性"""
        result = {}
        
        properties = self.parameters.get('properties', {})
        required = self.parameters.get('required', [])
        
        # 检查必填参数
        for req_param in required:
            if req_param not in params:
                raise ValidationError(f"Missing required parameter: {req_param}")
        
        # 验证每个参数
        for param_name, param_value in params.items():
            if param_name not in properties:
                raise ValidationError(f"Unknown parameter: {param_name}")
            
            prop_schema = properties[param_name]
            result[param_name] = self._validate_value(
                param_name, param_value, prop_schema
            )
        
        # 填充默认值
        for param_name, prop_schema in properties.items():
            if param_name not in result and 'default' in prop_schema:
                result[param_name] = prop_schema['default']
        
        return result
    
    def _validate_value(
        self, 
        name: str, 
        value: Any, 
        schema: Dict
    ) -> Any:
        """验证单个参数值"""
        param_type = schema.get('type')
        
        # 类型转换与校验
        if param_type == 'string':
            if not isinstance(value, str):
                value = str(value)
            # 检查 min/max length
            if 'minLength' in schema and len(value) < schema['minLength']:
                raise ValidationError(f"{name}: too short")
            if 'maxLength' in schema and len(value) > schema['maxLength']:
                raise ValidationError(f"{name}: too long")
                
        elif param_type == 'integer':
            try:
                value = int(value)
            except (ValueError, TypeError):
                raise ValidationError(f"{name}: must be integer")
            # 检查范围
            if 'minimum' in schema and value < schema['minimum']:
                raise ValidationError(f"{name}: below minimum {schema['minimum']}")
            if 'maximum' in schema and value > schema['maximum']:
                raise ValidationError(f"{name}: above maximum {schema['maximum']}")
                
        elif param_type == 'number':
            try:
                value = float(value)
            except (ValueError, TypeError):
                raise ValidationError(f"{name}: must be number")
                
        elif param_type == 'boolean':
            if isinstance(value, str):
                value = value.lower() in ('true', '1', 'yes', 'on')
            else:
                value = bool(value)
                
        elif param_type == 'array':
            if not isinstance(value, list):
                raise ValidationError(f"{name}: must be array")
            if 'minItems' in schema and len(value) < schema['minItems']:
                raise ValidationError(f"{name}: too few items")
            if 'maxItems' in schema and len(value) > schema['maxItems']:
                raise ValidationError(f"{name}: too many items")
                
        elif param_type == 'object':
            if not isinstance(value, dict):
                raise ValidationError(f"{name}: must be object")
        
        # 检查枚举值
        if 'enum' in schema and value not in schema['enum']:
            raise ValidationError(
                f"{name}: must be one of {schema['enum']}"
            )
        
        # 检查正则模式
        if 'pattern' in schema and isinstance(value, str):
            import re
            if not re.match(schema['pattern'], value):
                raise ValidationError(f"{name}: format invalid")
        
        return value

import time
import signal
from functools import wraps
from typing import Any, Callable


class SyncExecutor:
    """同步执行器，支持超时和重试"""
    
    def __init__(
        self,
        default_timeout: float = 30.0,
        max_retries: int = 3,
        retry_delay: float = 1.0
    ):
        self.default_timeout = default_timeout
        self.max_retries = max_retries
        self.retry_delay = retry_delay
    
    def execute(
        self,
        func: Callable,
        args: tuple = (),
        kwargs: dict = None,
        timeout: float = None
    ) -> Any:
        """
        执行函数，支持超时和重试
        
        Args:
            func: 要执行的函数
            args: 位置参数
            kwargs: 关键字参数
            timeout: 超时时间（秒）
            
        Returns:
            函数执行结果
            
        Raises:
            TimeoutError: 执行超时
            ExecutionError: 执行失败
        """
        timeout = timeout or self.default_timeout
        kwargs = kwargs or {}
        
        last_exception = None
        
        for attempt in range(self.max_retries + 1):
            try:
                # 使用信号实现超时（Unix）
                def timeout_handler(signum, frame):
                    raise TimeoutError(
                        f"Function execution timed out after {timeout}s"
                    )
                
                old_handler = signal.signal(signal.SIGALRM, timeout_handler)
                signal.alarm(int(timeout))
                
                try:
                    result = func(*args, **kwargs)
                    signal.alarm(0)  # 取消闹钟
                    return result
                finally:
                    signal.signal(signal.SIGALRM, old_handler)
                    
            except TimeoutError:
                raise  # 超时不再重试
            except Exception as e:
                last_exception = e
                if attempt < self.max_retries:
                    time.sleep(self.retry_delay * (2 ** attempt))
                    continue
                raise ExecutionError(
                    f"Function failed after {self.max_retries + 1} attempts: {e}"
                ) from last_exception

import asyncio
from concurrent.futures import ThreadPoolExecutor
from typing import List, Dict, Any


class AsyncExecutor:
    """异步执行器，支持并行工具调用"""
    
    def __init__(
        self,
        max_workers: int = 10,
        default_timeout: float = 30.0
    ):
        self.max_workers = max_workers
        self.default_timeout = default_timeout
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    async def execute_single(
        self,
        func: Callable,
        args: tuple = (),
        kwargs: dict = None,
        timeout: float = None
    ) -> Any:
        """执行单个工具"""
        kwargs = kwargs or {}
        timeout = timeout or self.default_timeout
        
        loop = asyncio.get_event_loop()
        
        try:
            return await asyncio.wait_for(
                loop.run_in_executor(
                    self.executor,
                    lambda: func(*args, **kwargs)
                ),
                timeout=timeout
            )
        except asyncio.TimeoutError:
            raise TimeoutError(f"Execution timed out after {timeout}s")
    
    async def execute_parallel(
        self,
        tasks: List[Dict[str, Any]]
    ) -> List[Any]:
        """
        并行执行多个工具
        
        Args:
            tasks: 任务列表，每个任务包含 func, args, kwargs
            
        Returns:
            结果列表（与 tasks 顺序一致）
        """
        coroutines = [
            self.execute_single(
                task['func'],
                task.get('args', ()),
                task.get('kwargs', {})
            )
            for task in tasks
        ]
        
        results = await asyncio.gather(
            *coroutines,
            return_exceptions=True
        )
        
        # 处理异常
        processed_results = []
        for i, result in enumerate(results):
            if isinstance(result, Exception):
                processed_results.append({
                    'error': str(result),
                    'task_index': i
                })
            else:
                processed_results.append(result)
        
        return processed_results


# 使用示例
async def main():
    executor = AsyncExecutor()
    
    # 并行执行多个搜索
    tasks = [
        {'func': web_search, 'kwargs': {'query': 'Apple 2025 revenue'}},
        {'func': web_search, 'kwargs': {'query': 'Google 2025 revenue'}},
        {'func': web_search, 'kwargs': {'query': 'Microsoft 2025 revenue'}}
    ]
    
    results = await executor.execute_parallel(tasks)
    return results

class ExecutionScheduler:
    """执行调度器，管理工具执行顺序和依赖"""
    
    def __init__(self, executor):
        self.executor = executor
    
    async def schedule_sequential(
        self,
        tasks: List[Dict]
    ) -> List[Any]:
        """顺序执行，前一个结果作为后一个输入"""
        results = []
        context = {}
        
        for task in tasks:
            # 注入上下文
            if 'context_injection' in task:
                for key, value_key in task['context_injection'].items():
                    task['kwargs'][key] = context.get(value_key)
            
            result = await self.executor.execute_single(
                task['func'],
                task.get('args', ()),
                task.get('kwargs', {})
            )
            
            results.append(result)
            
            # 更新上下文
            if 'context_output' in task:
                context[task['context_output']] = result
        
        return results
    
    async def schedule_dag(
        self,
        dag: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        基于 DAG 的执行调度
        
        dag = {
            'nodes': {
                'A': {'func': search_revenue, 'deps': []},
                'B': {'func': search_employees, 'deps': []},
                'C': {'func': calculate_ratio, 'deps': ['A', 'B']}
            }
        }
        """
        completed = {}
        pending = set(dag['nodes'].keys())
        
        while pending:
            # 找到所有依赖已完成的节点
            ready = [
                node_id for node_id in pending
                if all(
                    dep in completed
                    for dep in dag['nodes'][node_id]['deps']
                )
            ]
            
            if not ready:
                raise CircularDependencyError("Circular dependency detected")
            
            # 并行执行就绪节点
            tasks = [
                {
                    'func': dag['nodes'][node_id]['func'],
                    'kwargs': {
                        dep: completed[dep]
                        for dep in dag['nodes'][node_id]['deps']
                    }
                }
                for node_id in ready
            ]
            
            results = await self.executor.execute_parallel(tasks)
            
            for node_id, result in zip(ready, results):
                completed[node_id] = result
                pending.remove(node_id)
        
        return completed

from typing import Any, Dict, Optional
import json


class ResultProcessor:
    """工具结果处理器"""
    
    def __init__(
        self,
        max_length: int = 4000,
        truncation_strategy: str = 'smart'
    ):
        self.max_length = max_length
        self.truncation_strategy = truncation_strategy
    
    def process(
        self,
        result: Any,
        tool_name: str,
        execution_metadata: Dict = None
    ) -> str:
        """
        处理工具执行结果，转换为 LLM 可理解的格式
        
        Args:
            result: 原始结果（任意类型）
            tool_name: 工具名称
            execution_metadata: 执行元数据（耗时、状态等）
            
        Returns:
            格式化后的字符串
        """
        
        # 步骤1: 格式标准化
        formatted = self._format_result(result)
        
        # 步骤2: 截断控制
        truncated = self._truncate(formatted)
        
        # 步骤3: 包装为 Observation 格式
        observation = self._wrap_observation(
            truncated, tool_name, execution_metadata
        )
        
        return observation
    
    def _format_result(self, result: Any) -> str:
        """将任意结果转换为字符串"""
        if result is None:
            return "No result returned."
        
        if isinstance(result, str):
            return result
        
        if isinstance(result, (list, dict)):
            return json.dumps(result, ensure_ascii=False, indent=2)
        
        if isinstance(result, Exception):
            return f"Error: {str(result)}"
        
        return str(result)
    
    def _truncate(self, text: str) -> str:
        """截断长文本"""
        if len(text) <= self.max_length:
            return text
        
        if self.truncation_strategy == 'head':
            return text[:self.max_length] + "... [truncated]"
        
        elif self.truncation_strategy == 'tail':
            return "[truncated] ..." + text[-self.max_length:]
        
        elif self.truncation_strategy == 'smart':
            # 智能截断：保留开头和结尾，中间省略
            half = self.max_length // 2
            return (
                text[:half] + 
                f"\n\n... [truncated {len(text) - self.max_length} characters] ...\n\n" + 
                text[-half:]
            )
        
        return text[:self.max_length]
    
    def _wrap_observation(
        self,
        content: str,
        tool_name: str,
        metadata: Dict = None
    ) -> str:
        """包装为 Observation 格式"""
        metadata = metadata or {}
        
        # 基础格式
        observation = f"Observation [{tool_name}]:\n{content}"
        
        # 添加元数据（可选）
        if metadata.get('execution_time'):
            observation += f"\n\n[Execution time: {metadata['execution_time']:.2f}s]"
        
        if metadata.get('status'):
            observation += f"\n[Status: {metadata['status']}]"
        
        return observation

        permissions: List[ToolPermission]
    ):
        """定义角色权限"""
        self.roles[role_name] = {
            'permissions': {p.tool_name: p for p in permissions}
        }
    
    def assign_role(self, user_id: str, role_name: str):
        """为用户分配角色"""
        self.user_roles[user_id] = role_name
    
    def check_permission(
        self,
        user_id: str,
        tool_name: str,
        params: Dict
    ) -> bool:
        """检查用户是否有权限执行工具"""
        role_name = self.user_roles.get(user_id)
        if not role_name:
            return False
        
        role = self.roles.get(role_name)
        if not role:
            return False
        
        permission = role['permissions'].get(tool_name)
        if not permission:
            return False
        
        # 检查参数权限
        for param_name in params:
            if permission.forbidden_params and param_name in permission.forbidden_params:
                return False
            
            if permission.allowed_params and param_name not in permission.allowed_params:
                return False
        
        return True

9. MCP 协议详解

9.1 什么是 MCP

MCP (Model Context Protocol) 是 Anthropic 于 2024 年底推出的开放标准协议，旨在统一 AI Agent 与外部工具、数据源、服务的接口。到 2026 年，MCP 已成为 Agent 生态的事实标准。

核心定位: MCP 是 Agent 世界的 "HTTP" - 它定义了通用的通信协议，使得任何支持 MCP 的客户端可以无缝接入任何 MCP Server。

import json
import subprocess
from typing import List, Dict, Any


class MCPClient:
    """MCP 协议客户端"""
    
    def __init__(self, server_command: List[str]):
        self.server_command = server_command
        self.process = None
        self.tools = []
    
    def connect(self):
        """启动 MCP Server 进程"""
        self.process = subprocess.Popen(
            self.server_command,
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True
        )
        
        # 初始化握手
        self._send({
            'jsonrpc': '2.0',
            'id': 0,
            'method': 'initialize',
            'params': {
                'protocolVersion': '2024-11-05',
                'capabilities': {},
                'clientInfo': {
                    'name': 'example-client',
                    'version': '1.0.0'
                }
            }
        })
        
        response = self._receive()
        if 'error' in response:
            raise ConnectionError(f"MCP init failed: {response['error']}")
    
    def discover_tools(self) -> List[Dict]:
        """发现可用工具"""
        self._send({
            'jsonrpc': '2.0',
            'id': 1,
            'method': 'tools/list',
            'params': {}
        })
        
        response = self._receive()
        self.tools = response.get('result', {}).get('tools', [])
        return self.tools
    
    def call_tool(
        self,
        tool_name: str,
        arguments: Dict[str, Any]
    ) -> Dict:
        """调用工具"""
        self._send({
            'jsonrpc': '2.0',
            'id': 2,
            'method': 'tools/call',
            'params': {
                'name': tool_name,
                'arguments': arguments
            }
        })
        
        return self._receive()
    
    def _send(self, message: Dict):
        """发送消息到 MCP Server"""
        json_line = json.dumps(message) + '\n'
        self.process.stdin.write(json_line)
        self.process.stdin.flush()
    
    def _receive(self) -> Dict:
        """接收 MCP Server 响应"""
        line = self.process.stdout.readline()
        return json.loads(line)
    
    def close(self):
        """关闭连接"""
        if self.process:
            self.process.terminate()
            self.process.wait()


# 使用示例
client = MCPClient(['python', 'mcp_server.py'])
client.connect()
tools = client.discover_tools()
result = client.call_tool('web_search', {'query': 'Apple revenue 2025'})
client.close()

9.6 MCP 与 A2A 的关系

xxxxxxxxxx
MCP (Model Context Protocol)
    |-- Agent <-> 工具/数据（垂直集成）
    |-- 解决：Agent 如何获取能力
    |-- 类比：HTTP 协议（请求-响应）
    
A2A (Agent-to-Agent Protocol)
    |-- Agent <-> Agent（水平协作）
    |-- 解决：Agent 如何协作完成任务
    |-- 类比：WebSocket（双向通信）
    
两者互补：
    MCP 让 Agent 获取能力 --> A2A 让 Agent 协作完成任务

import hashlib
import time
from typing import Any, Dict, Optional


class ToolResultCache:
    """工具结果缓存系统"""
    
    def __init__(
        self,
        default_ttl: int = 300,  # 5分钟
        max_size: int = 1000
    ):
        self.default_ttl = default_ttl
        self.max_size = max_size
        self.cache = {}  # key -> (value, timestamp, ttl)
    
    def _generate_key(
        self,
        tool_name: str,
        arguments: Dict
    ) -> str:
        """生成缓存键"""
        key_data = f"{tool_name}:{json.dumps(arguments, sort_keys=True)}"
        return hashlib.md5(key_data.encode()).hexdigest()
    
    def get(
        self,
        tool_name: str,
        arguments: Dict
    ) -> Optional[Any]:
        """获取缓存结果"""
        key = self._generate_key(tool_name, arguments)
        
        if key not in self.cache:
            return None
        
        value, timestamp, ttl = self.cache[key]
        
        # 检查是否过期
        if time.time() - timestamp > ttl:
            del self.cache[key]
            return None
        
        return value
    
    def set(
        self,
        tool_name: str,
        arguments: Dict,
        value: Any,
        ttl: int = None
    ):
        """设置缓存"""
        # 清理过期缓存
        self._cleanup()
        
        # 限制缓存大小
        if len(self.cache) >= self.max_size:
            self._evict_oldest()
        
        key = self._generate_key(tool_name, arguments)
        self.cache[key] = (value, time.time(), ttl or self.default_ttl)
    
    def _cleanup(self):
        """清理过期缓存"""
        now = time.time()
        expired = [
            k for k, (_, ts, ttl) in self.cache.items()
            if now - ts > ttl
        ]
        for k in expired:
            del self.cache[k]
    
    def _evict_oldest(self):
        """淘汰最旧的缓存"""
        if not self.cache:
            return
        oldest = min(self.cache.keys(), key=lambda k: self.cache[k][1])
        del self.cache[oldest]

    async def execute_parallel(
        self,
        tool_calls: List[Dict]
    ) -> List[Tuple[str, Any]]:
        """
        并行执行多个工具调用
        
        Args:
            tool_calls: [{'tool': func, 'name': str, 'args': dict}, ...]
            
        Returns:
            [(tool_name, result), ...]
        """
        
        async def _execute_single(tool_call):
            async with self.semaphore:
                try:
                    result = await tool_call['tool'](**tool_call['args'])
                    return (tool_call['name'], result)
                except Exception as e:
                    return (tool_call['name'], {'error': str(e)})
        
        tasks = [_execute_single(tc) for tc in tool_calls]
        return await asyncio.gather(*tasks)

        max_length: int = 1024

        优化工具描述以提升 LLM 理解

        1. 明确使用场景（When to use）
        2. 明确输入格式（What to provide）
        3. 明确输出格式（What to expect）
        4. 提供示例（Example）
        5. 避免歧义（Avoid ambiguity）
        """
        
        # 检查描述质量
        checks = self._quality_check(description)
        
        optimized = description
        
        # 添加使用场景提示
        if not checks['has_when_to_use']:
            optimized = "Use this tool when you need to... " + optimized
        
        # 截断到最大长度
        if len(optimized) > max_length:
            optimized = optimized[:max_length-3] + '...'
        
        return optimized
    
    def _quality_check(self, description: str) -> Dict[str, bool]:
        """检查描述质量"""
        return {
            'has_when_to_use': 'when' in description.lower(),
            'has_example': 'example' in description.lower(),
            'has_input_hint': 'input' in description.lower() or 'parameter' in description.lower(),
            'has_output_hint': 'return' in description.lower() or 'output' in description.lower(),
            'not_too_short': len(description) > 50,
            'not_too_long': len(description) < 1024
        }

class ToolCallTracer:
    """工具调用追踪器"""
    
    def __init__(self):
        self.records: List[ToolCallRecord] = []
        self.current_session: Optional[str] = None
    
    def start_trace(self, session_id: str):
        """开始追踪会话"""
        self.current_session = session_id
        self.records = []
    
    def log_call(
        self,
        tool_name: str,
        arguments: Dict
    ) -> ToolCallRecord:
        """记录工具调用开始"""
        record = ToolCallRecord(
            tool_name=tool_name,
            arguments=arguments,
            start_time=time.time()
        )
        self.records.append(record)
        return record
    
    def log_result(
        self,
        record: ToolCallRecord,
        result: Any,
        error: Exception = None
    ):
        """记录工具调用结果"""
        record.end_time = time.time()
        record.result = result
        if error:
            record.error = str(error)
    
    def get_summary(self) -> Dict:
        """获取追踪摘要"""
        total_calls = len(self.records)
        successful = sum(1 for r in self.records if r.error is None)
        failed = total_calls - successful
        
        total_time = sum(
            (r.end_time or r.start_time) - r.start_time
            for r in self.records
        )
        
        tool_usage = {}
        for r in self.records:
            if r.tool_name not in tool_usage:
                tool_usage[r.tool_name] = {'count': 0, 'total_time': 0, 'errors': 0}
            tool_usage[r.tool_name]['count'] += 1
            tool_usage[r.tool_name]['total_time'] += (r.end_time or r.start_time) - r.start_time
            if r.error:
                tool_usage[r.tool_name]['errors'] += 1
        
        return {
            'session_id': self.current_session,
            'total_calls': total_calls,
            'successful': successful,
            'failed': failed,
            'success_rate': successful / total_calls if total_calls > 0 else 0,
            'total_time': total_time,
            'avg_time': total_time / total_calls if total_calls > 0 else 0,
            'tool_usage': tool_usage
        }

# 最小可运行的工具调用系统
from typing import Annotated
from pydantic import Field


class SimpleToolSystem:
    def __init__(self):
        self.tools = {}
    
    def register(self, func):
        self.tools[func.__name__] = func
        return func
    
    def call(self, name, **kwargs):
        if name not in self.tools:
            raise ValueError(f"Tool {name} not found")
        return self.tools[name](**kwargs)


system = SimpleToolSystem()


@system.register
def search(query: Annotated[str, Field(description="搜索查询")]):
    """搜索工具"""
    return f"搜索结果: {query}"


@system.register
def calculate(expression: Annotated[str, Field(description="数学表达式")]):
    """计算工具"""
    return eval(expression)


# 使用
print(system.call('search', query='Apple revenue'))
print(system.call('calculate', expression='2 + 2'))
